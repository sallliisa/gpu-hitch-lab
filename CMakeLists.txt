cmake_minimum_required(VERSION 3.21)
project(GpuHitchLab LANGUAGES CXX)

function(configure_gpu_hitch_target target)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  target_link_libraries(${target} PRIVATE d3d12 dxgi dxguid winmm shell32)
endfunction()

add_executable(GpuHitchLab WIN32 src/main.cpp)
configure_gpu_hitch_target(GpuHitchLab)

option(GPU_HITCH_BUILD_REFLEX "Build NVIDIA Reflex edition (requires NVIDIA NVAPI SDK)" OFF)
if(GPU_HITCH_BUILD_REFLEX)
  set(NVAPI_ROOT "${CMAKE_SOURCE_DIR}/third_party/nvapi")
  if(NOT EXISTS "${NVAPI_ROOT}/nvapi.h" OR NOT EXISTS "${NVAPI_ROOT}/amd64/nvapi64.lib")
    message(FATAL_ERROR "NVIDIA NVAPI SDK missing at third_party/nvapi")
  endif()

  add_executable(GpuHitchLabReflex WIN32 src/main.cpp)
  configure_gpu_hitch_target(GpuHitchLabReflex)
  target_compile_definitions(GpuHitchLabReflex PRIVATE GPU_HITCH_LAB_REFLEX=1)
  target_include_directories(GpuHitchLabReflex PRIVATE "${NVAPI_ROOT}")
  target_link_libraries(GpuHitchLabReflex PRIVATE "${NVAPI_ROOT}/amd64/nvapi64.lib")
endif()
